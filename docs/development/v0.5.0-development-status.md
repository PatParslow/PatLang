#### AST Node Hierarchy Extensions
```ruby
class FunctionDefinitionNode < ASTNode
  attr_reader :name, :parameters, :body, :return_type
  
  def initialize(name, parameters, body, return_type = nil)
    @name = name
    @parameters = parameters || []
    @body = body
    @return_type = return_type
  end
end

class FunctionCallNode < ASTNode
  attr_reader :function_name, :arguments
  
  def initialize(function_name, arguments = [])
    @function_name = function_name
    @arguments = arguments || []
  end
end

class ParameterNode < ASTNode
  attr_reader :name, :type, :default_value
  
  def initialize(name, type = nil, default_value = nil)
    @name = name
    @type = type
    @default_value = default_value
  end
end

class ReturnNode < ASTNode
  attr_reader :expression
  
  def initialize(expression = nil)
    @expression = expression
  end
end
```

#### Lexer Natural Language Enhancement
```ruby
# Enhanced natural language phrase detection
def scan_function_phrases
  if @current_char == 'm' && peek_ahead('make a function called')
    advance_by('make a function called'.length)
    return Token.new(TokenType::FUNCTION_DEF, 'make a function called', @line, @column)
  end
  
  if match_word('takes') && peek_next_char == ':'
    advance_by('takes'.length)
    return Token.new(TokenType::TAKES, 'takes', @line, @column)
  end
  
  if match_word('returns') && peek_next_char == ':'
    advance_by('returns'.length)
    return Token.new(TokenType::RETURNS, 'returns', @line, @column)
  end
end
```
