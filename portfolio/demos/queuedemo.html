Durable message queue: a crash, and a real recovery

Three independent workers, each doing risky I/O work, each with its OWN topic (queue_writer_topic(base, worker_id)) rather than sharing one — the queue does whole-file read-modify-write with no locking, so giving every writer its own topic avoids a write race by construction, not by coordination. Two workers finish cleanly via queue_attempt (publish intent, do the work, ack the outcome). The third genuinely crashes mid-attempt — PatLang has no try/catch, so a real fatal host error (a bad read_file call, verified this session across four actually-separate OS processes) kills that worker outright, before it ever reaches queue_ack. A separate, later process then calls queue_pending_multi across all three workers' topics, finds the one leftover checkpoint, completes it, and acks it — proving the intent genuinely survived the crash on disk, not just in this transcript. Built entirely in self_hosting/lib/queue.patlang, no new Rust host functions. See the Paradigms Guide's Durable Message Queue section for the full design. Native transcript only, no "run in browser" button: needs real file I/O and a real fatal process exit, neither available in the WASI sandbox. The transcript below represents the crash directly (publishing the intent without ever calling the action or the ack — exactly the on-disk state a real crash between those two steps leaves) since a genuine fatal error can't be captured mid-transcript without killing this whole build — see self_hosting/examples/queue_recovery_demo.patlang for the real, actually-run-it-yourself four-separate-process form.

# Self-hosted mirror of rust-runtime/src/preprocess.rs's expand_includes:
# expands `include "relative/path.patlang"` lines by splicing the
# referenced file's contents in place, resolving paths relative to the
# including file's own directory, recursively.
#
# Why this exists as PatLang, not just Rust: `expand_includes` was
# previously a NATIVE-ONLY preprocessing step (main.rs, run before the
# frontend ever sees the source) -- patc1.exe's own self-hosted lexer/
# parser never learned to do this, so any .patlang file using `include`
# could only be compiled via the native pat.exe frontend (`--ir-run`/
# `--patc`), never handed directly to patc1.exe, which is why every
# multi-file portfolio demo in build_portfolio.patlang manually
# concatenates dependency files (read_file(lexer) + chr(10) + ...) instead
# of using `include`. This closes that gap so `include` works identically
# everywhere -- interpreted, natively compiled, and self-hosted-compiled --
# matching this session's usual bar of "verified across all three paths."
#
# Kept as its own small library (not folded directly into patc1_main.patlang)
# so any self-hosted driver can `include "lib/includes.patlang"` and use it.

# str_trim(s) -> s with leading/trailing space/tab/\r/\n stripped.
# GitHub #22: \n was deliberately excluded here originally; audited every
# call site before changing this shared utility's semantics (per the
# issue's own explicit request not to "fix" it without checking callers
# first). Every current caller either trims an already-line-split string
# (no embedded \n to lose) or explicitly WANTS trailing newlines stripped
# (the two run_benchmarks.patlang/webcrawler.patlang callers comparing a
# captured-output blob across execution paths -- the original bug report
# that surfaced this: str_trim() alone didn't close a trailing-newline
# difference, needing a separate local helper to finish the job). No
# caller relies on \n being preserved through a trim call.
make a function called str_trim takes s returns trimmed
  let n = s.length
  let start = 0
  while (start < n) and is_ws_char(char_code(s, start)) do
    let start = start + 1
  end
  let end = n
  while (end > start) and is_ws_char(char_code(s, end - 1)) do
    let end = end - 1
  end
  return substr(s, start, end - start)
end

make a function called is_ws_char takes code returns is_ws
  return (code == 32) or (code == 9) or (code == 13) or (code == 10)
end

# str_starts_with(s, prefix) -> bool
make a function called str_starts_with takes s, prefix returns matches
  if prefix.length > s.length then
    return false
  end
  return substr(s, 0, prefix.length) == prefix
end

# str_ends_with(s, suffix) -> bool
make a function called str_ends_with takes s, suffix returns matches
  if suffix.length > s.length then
    return false
  end
  return substr(s, s.length - suffix.length, suffix.length) == suffix
end

# split_lines(s) -> list of lines, split on \n (a trailing \r on each line,
# from CRLF source files, is stripped too).
#
# char_code and substr each check is_ascii() over their WHOLE string
# argument on every call (hosts.rs and codegen.rs's runtime alike), so a
# per-character loop over `s` itself is O(n^2). That made assembling one
# 6 MB function (x64_assemble_chunk splits its whole asm text here) take
# ~20 minutes. Instead `s` is first cut into pieces of at most 1024
# characters by repeated halving (each level of halving costs O(n), so
# O(n log n) overall), and only the short pieces are scanned character by
# character. A line spanning pieces is carried as a list of segments and
# joined in a string builder only then (builders are never freed, so one
# per line would leak; this is at most one per 1024 characters).
# Checked by self_hosting/tools/split_lines_check.patlang.
make a function called split_lines takes s returns lines
  let pieces = split_lines_pieces(s, [])
  let out = []
  let carry = []
  let pi = 0
  let pn = list_len(pieces)
  while pi < pn do
    let piece = list_get(pieces, pi)
    let m = piece.length
    let start = 0
    let i = 0
    while i < m do
      if char_code(piece, i) == 10 then
        let seg = substr(piece, start, i - start)
        if list_len(carry) > 0 then
          let seg = split_lines_join(list_push(carry, seg))
          let carry = []
        end
        let out = list_push(out, strip_trailing_cr(seg))
        let start = i + 1
      end
      let i = i + 1
    end
    if start < m then
      let carry = list_push(carry, substr(piece, start, m - start))
    end
    let pi = pi + 1
  end
  if list_len(carry) > 0 then
    let out = list_push(out, strip_trailing_cr(split_lines_join(carry)))
  end
  return out
end

make a function called split_lines_join takes segs returns joined
  let b = sb_new()
  let i = 0
  while i < list_len(segs) do
    sb_push(b, list_get(segs, i))
    let i = i + 1
  end
  return sb_str(b)
end

# Appends `s` to `out` as consecutive pieces of at most 1024 characters,
# splitting at the largest power-of-two multiple of 1024 below its length.
make a function called split_lines_pieces takes s, out returns out2
  let n = s.length
  if n <= 1024 then
    return list_push(out, s)
  end
  let p = 1024
  while (p + p) < n do
    let p = p + p
  end
  let out = split_lines_pieces(substr(s, 0, p), out)
  return split_lines_pieces(substr(s, p, n - p), out)
end

make a function called strip_trailing_cr takes line returns stripped
  let n = line.length
  if (n > 0) and (char_code(line, n - 1) == 13) then
    return substr(line, 0, n - 1)
  end
  return line
end

# path_dirname(path) -> everything before the last '/' or '\', or "." if
# the path has no directory component. Handles both separators since
# build_portfolio.patlang and friends run on Windows but write forward
# slashes in string literals.
make a function called path_dirname takes path returns dir
  let n = path.length
  let i = n - 1
  let last_sep = -1
  while i >= 0 do
    let c = char_code(path, i)
    if (c == 47) or (c == 92) then
      let last_sep = i
      let i = -1
    else
      let i = i - 1
    end
  end
  if last_sep < 0 then
    return "."
  end
  return substr(path, 0, last_sep)
end

# path_basename(path) -> everything after the last '/' or '\', or the
# whole path if it has no directory component -- the complement of
# path_dirname above (same separator-scanning loop, opposite half kept).
make a function called path_basename takes path returns base
  let n = path.length
  let i = n - 1
  let last_sep = -1
  while i >= 0 do
    let c = char_code(path, i)
    if (c == 47) or (c == 92) then
      let last_sep = i
      let i = -1
    else
      let i = i - 1
    end
  end
  if last_sep < 0 then
    return path
  end
  return substr(path, last_sep + 1, n - last_sep - 1)
end

# path_join(base, rel) -> base + "/" + rel, tolerating a trailing slash on
# base and an empty base (meaning "current directory"). If `rel` is itself
# absolute (leading '/'/'\', or a Windows drive letter like "C:"), it's
# returned unchanged, ignoring base -- matches Rust's PathBuf::join, which
# preprocess.rs's native expand_includes relies on for the same case.
make a function called path_join takes base, rel returns joined
  if is_absolute_path(rel) then
    return rel
  end
  if (base == "") or (base == ".") then
    return rel
  end
  let n = base.length
  if (n > 0) and ((char_code(base, n - 1) == 47) or (char_code(base, n - 1) == 92)) then
    return base + rel
  end
  return base + "/" + rel
end

make a function called is_absolute_path takes p returns is_abs
  if p.length == 0 then
    return false
  end
  let c0 = char_code(p, 0)
  if (c0 == 47) or (c0 == 92) then
    return true
  end
  if (p.length >= 2) and (char_code(p, 1) == 58) then
    return true
  end
  return false
end

# expand_includes(source, base_dir) -> source with every `include "path"`
# line recursively replaced by that file's own (recursively expanded)
# contents, paths resolved relative to base_dir (the including file's own
# directory) at each level, exactly matching preprocess.rs's semantics.
#
# The depth cap (16, matching preprocess.rs's MAX_DEPTH) is inlined as a
# literal below rather than a top-level `let` constant referenced from
# inside expand_includes_at_depth -- patc1.exe was found, while building
# this, to NOT make top-level `let` constants visible inside function
# bodies at all (confirmed via a minimal repro: the value silently reads
# as empty/unset, not an error) even though both --ir-run and native
# --patc handle this correctly. That's a real, previously-unknown
# self-hosted-compiler bug, logged separately in the backlog for its own
# dedicated fix -- this file just avoids relying on the broken behavior.
make a function called expand_includes takes source, base_dir returns expanded
  return expand_includes_at_depth(source, base_dir, 0)
end

# expand_includes with each file once, however many files include it, as a compiler
# reads them (docs/spec/values-and-memory.md 1.13).
make a function called expand_includes_once takes source, base_dir returns expanded
  return expand_includes_once_with(source, base_dir, vec_new())
end

make a function called expand_includes_once_with takes source, base_dir, seen returns expanded
  let lines = split_lines(source)
  let out = sb_new()
  let i = 0
  while i < list_len(lines) do
    let t = str_trim(lines[i])
    if str_starts_with(t, "include ") then
      let path = path_join(base_dir, strip_quotes(str_trim(substr(t, 8, t.length - 8))))
      let key = path_normalize(path)
      let known = false
      let k = 0
      while k < vec_len(seen) do
        if vec_get(seen, k) == key then
          let known = true
        end
        let k = k + 1
      end
      if not known then
        vec_push(seen, key)
        sb_push(out, expand_includes_once_with(read_file(path), path_dirname(path), seen))
      end
    else
      sb_push(out, lines[i])
    end
    sb_push(out, chr(10))
    let i = i + 1
  end
  return sb_str(out)
end

# path with "." segments dropped and each "seg/.." pair removed, "/" between
# segments ("a/b/../c" -> "a/c"; a leading ".." that has nothing to cancel is
# kept), so an include reached as "spec_fixtures/../heap.patlang" names its
# real directory.
make a function called path_normalize takes path returns out
  let segs = []
  let cur = sb_new()
  let n = path.length
  let i = 0
  while i <= n do
    let c = 47
    if i < n then
      let c = char_code(path, i)
    end
    if (c == 47) or (c == 92) then
      let seg = sb_str(cur)
      let cur = sb_new()
      if seg == ".." then
        let k = list_len(segs)
        if (k > 0) and (segs[k - 1] != "..") and (segs[k - 1] != "") then
          let kept = []
          let j = 0
          while j < k - 1 do
            let kept = list_push(kept, segs[j])
            let j = j + 1
          end
          let segs = kept
        else
          let segs = list_push(segs, seg)
        end
      elif (seg != ".") and ((seg != "") or (list_len(segs) == 0)) then
        let segs = list_push(segs, seg)
      end
    else
      sb_push(cur, chr(c))
    end
    let i = i + 1
  end
  let b = sb_new()
  let k = 0
  while k < list_len(segs) do
    if k > 0 then
      sb_push(b, "/")
    end
    sb_push(b, segs[k])
    let k = k + 1
  end
  return sb_str(b)
end

make a function called expand_includes_at_depth takes source, base_dir, depth returns expanded
  return expand_includes_at_depth_opts(source, base_dir, depth, false)
end

make a function called expand_includes_at_depth_opts takes source, base_dir, depth, native returns expanded
  if depth > 16 then
    print("include: nesting deeper than 16 levels (cycle?)")
    return source
  end
  let lines = split_lines(source)
  let out = sb_new()
  let i = 0
  let n = to_num(list_len(lines))
  while i < n do
    let line = lines[i]
    let t = str_trim(line)
    if str_starts_with(t, "include ") and (str_starts_with(t, "#") == false) then
      let rel = str_trim(substr(t, 8, t.length - 8))
      let rel = strip_quotes(rel)
      let path = path_join(base_dir, rel)
      if native and x64_runtime_bundled_source(path) then
        sb_push(out, chr(10))
      else
        let inner = read_file(path)
        let inner_base = path_dirname(path)
        sb_push(out, expand_includes_at_depth_opts(inner, inner_base, depth + 1, native))
        sb_push(out, chr(10))
      end
    else
      sb_push(out, line)
      sb_push(out, chr(10))
    end
    let i = i + 1
  end
  return sb_str(out)
end

# strip_quotes("\"path\"") -> "path" -- include lines are always written
# with double-quoted paths, same as the native preprocessor expects.
make a function called strip_quotes takes s returns unquoted
  let n = s.length
  if (n >= 2) and (char_code(s, 0) == 34) and (char_code(s, n - 1) == 34) then
    return substr(s, 1, n - 2)
  end
  return s
end

# A durable, file-backed message queue -- entirely PatLang, no new Rust host
# functions. One key/file PER MESSAGE (self_hosting/lib/mailbox.patlang's own
# layout, independently reimplemented here rather than included -- queue.patlang
# is included from files that already include mailbox.patlang transitively
# [e.g. via isolate.patlang], and this codebase's include mechanism has no
# dedup, so a second copy of the SAME functions would collide; a few small
# local helpers (q_pad etc.) cost far less than that risk):
#   fs driver:  q_dir()/<topic>/<padded_id>         (plus .../.seq, the next-id counter)
#   vfs driver: /queue/<topic>/<padded_id>          (plus .../.seq)
# Each message's own file/key holds "status<TAB>payload". Originally one flat
# log file per topic (q_dir()/<topic>.log, tab-separated id/status/payload
# lines, rewritten WHOLE on every publish/ack/consume) -- GitHub #120: that
# shape meant a crash mid-write during any one call could corrupt or lose
# EVERY message in the topic, not just the one being touched. Per-message
# storage means a crash only ever risks the one message actually being
# written; every other already-committed message's own file is untouched.
# This does NOT add new protection against two CONCURRENT writers to the SAME
# topic racing on the same logical operation (the id-allocation counter is
# still a single read-then-write value, same residual limitation mailbox.
# patlang's own header honestly documents for its own counter) -- the
# established mitigation for that, unchanged, is still queue_writer_topic:
# give each writer its own topic so no two writers ever touch the same
# message set at all.
#
# An existing topic in the OLD flat-log shape is migrated the first time
# ANY call touches it (q_migrate_if_old): its rows are read once, written
# into the new per-message layout, and the old file/key is removed -- so a
# live system upgrading across this change loses nothing already durably
# published, and every topic ends up in exactly one place after its first
# touch post-upgrade, not two formats checked forever.
#
# The saga/contract-shaped wrapper (queue_attempt) realizes the pattern:
# intent = require, attempt = do the risky thing, outcome = ensure. If the
# wrapped function aborts (PatLang has no try/catch -- a host error is
# still fatal, unchanged by this library), the intent message is simply
# left "pending" on disk for a later run (or a dedicated recovery pass) to
# find via queue_pending and decide whether to retry or compensate.
#
# Storage driver: the log lives in real files ("fs", durable across
# processes) wherever host_caps() lists "fs", and in the in-memory VFS
# ("vfs", every target including the browser, gone when the world ends)
# where it does not. queue_use_driver("vfs") or queue_use_driver("fs")
# forces one; every queue_* call behaves the same on either.
#
# NOTE on style: q_dir() is a function, not a top-level `let`, for the same
# reason build_daemon.patlang's "constants" all are -- see that file's own
# header comment and [[patlang-backlog]] item 10.

# "fs" or "vfs": the forced driver if queue_use_driver was called, else "fs"
# when this runtime has files and "vfs" when it does not.
make a function called queue_driver returns name
  if vfs_exists(q_driver_key()) == "1" then return vfs_read(q_driver_key()) end
  let caps = host_caps()
  let i = 0
  while i < list_len(caps) do
    if caps[i] == "fs" then return "fs" end
    let i = i + 1
  end
  return "vfs"
end

# Split an OLD-format flat-log line into [id, status, payload]. Only used
# by the one-time migration read now (q_read_old_fs/q_read_old_vfs);
# everything else goes through the new per-message layout directly.
make a function called q_parse_line takes line returns fields
  let t = q_tab()
  let first = find_substr_queue(line, t)
  if first < 0 then
    return ["", "", ""]
  end
  let rest = substr(line, first + 1, line.length - first - 1)
  let second = find_substr_queue(rest, t)
  if second < 0 then
    return [substr(line, 0, first), rest, ""]
  end
  let id = substr(line, 0, first)
  let status = substr(rest, 0, second)
  let payload = substr(rest, second + 1, rest.length - second - 1)
  return [id, status, payload]
end

make a function called find_substr_queue takes hay, needle returns idx
  let hn = hay.length
  let nn = needle.length
  if nn == 0 then
    return 0
  end
  let i = 0
  while i <= hn - nn do
    if substr(hay, i, nn) == needle then
      return i
    end
    let i = i + 1
  end
  return -1
end

make a function called q_unpack_status_payload takes text returns pair
  let sep = find_substr_queue(text, q_tab())
  if sep < 0 then
    return ["", text]
  end
  return [substr(text, 0, sep), substr(text, sep + 1, text.length - sep - 1)]
end

make a function called q_get_counter takes topic returns next_id

    if vfs_exists(q_counter_key(topic)) == "1" then

    end
    return 0
  end

    return to_num(read_file(q_counter_path(topic)))
  end

end

  end

make a function called q_put_message takes topic, id, status, payload returns done

  if queue_driver() == "vfs" then

    write_file(q_msg_dir(topic) + "/" + q_pad(id), text)
  end
  return true
end

make a function called q_delete_message takes topic, padded_id returns done

  end
  return true
end

# Padded message ids for a topic, oldest first (vfs_list/list_dir are both
# already sorted, and equal-width zero-padded ids sort identically as
# strings or numbers).
make a function called q_message_ids takes topic returns ids
  if queue_driver() == "vfs" then
    let prefix = q_msg_prefix(topic)
    let keys = vfs_list(prefix)
    let out = []
    let i = 0
    while i < list_len(keys) do
      let suf = substr(keys[i], prefix.length, keys[i].length - prefix.length)
      if suf != ".seq" then
        let out = list_push(out, suf)
      end
      let i = i + 1
    end
    return out
  end
  if file_exists(q_msg_dir(topic)) != true then
    return []
  end
  let entries = list_dir(q_msg_dir(topic))
  let out = []
  let i = 0
  while i < list_len(entries) do
    if entries[i] != ".seq" then
      let out = list_push(out, entries[i])
    end
    let i = i + 1
  end
  return out
end

make a function called q_list_contains takes xs, v returns found

end

  if file_exists(q_file(topic)) != true then
    return []
  end
  return q_parse_raw(read_file(q_file(topic)))
end

make a function called q_read_old_vfs takes topic returns rows
  if vfs_exists(q_key(topic)) != "1" then
    return []
  end
  return q_parse_raw(vfs_read(q_key(topic)))
end

# Writes a full [id, status, payload] row list into the new per-message
# layout from scratch, and advances the topic's counter past the highest
# id written (never backward). Used by migration and by q_write below.
make a function called q_write_rows_new takes topic, rows returns done
  let i = 0
  let n = to_num(list_len(rows))
  let max_id = -1
  while i < n do
    let row = rows[i]
    let id = to_num(row[0])
    if id > max_id then
      let max_id = id
    end
    q_put_message(topic, id, row[1], row[2])
    let i = i + 1
  end
  if max_id + 1 > q_get_counter(topic) then
    q_set_counter(topic, max_id + 1)
  end
  return true
end

make a function called q_migrate_if_old takes topic returns done
  if queue_driver() == "vfs" then
    if vfs_exists(q_key(topic)) == "1" then
      q_write_rows_new(topic, q_read_old_vfs(topic))
      vfs_delete(q_key(topic))
    end
  else
    if file_exists(q_file(topic)) == true then
      q_write_rows_new(topic, q_read_old_fs(topic))
      remove_file(q_file(topic))
    end
  end
  return true
end

# ---- public read/write (queue_publish/consume/ack/pending, and
# task_registry.patlang's own direct calls, are all built on just these
# two -- unchanged contract from the caller's point of view) ----

# Read a topic's rows as [id, status, payload] triples, oldest first.
# Returns [] if the topic has no messages yet.
make a function called q_read takes topic returns rows
  q_migrate_if_old(topic)
  let ids = q_message_ids(topic)
  let out = []
  let i = 0
  let n = to_num(list_len(ids))
  while i < n do
    let padded = ids[i]
    let unpacked = q_unpack_status_payload(q_read_message(topic, padded))
    let out = list_push(out, ["" + to_num(padded), unpacked[0], unpacked[1]])
    let i = i + 1
  end
  return out
end

# Replaces a topic's entire content with exactly these rows -- any
# existing message not present in `rows` is removed. This is the exact
# semantics task_registry.patlang's own direct q_write(topic, [[0, ...]])
# call relies on (the presence-beacon idiom: "this topic holds only its
# latest state"), and what queue_ack below reconstructs internally.
make a function called q_write takes topic, rows returns done
  q_migrate_if_old(topic)
  let existing = q_message_ids(topic)
  let keep = []
  let i = 0
  let n = to_num(list_len(rows))
  while i < n do
    let row = rows[i]
    let id = to_num(row[0])
    q_put_message(topic, id, row[1], row[2])
    let keep = list_push(keep, q_pad(id))
    let i = i + 1
  end
  let j = 0
  let en = to_num(list_len(existing))
  while j < en do
    if q_list_contains(keep, existing[j]) == false then
      q_delete_message(topic, existing[j])
    end
    let j = j + 1
  end
  let max_id = -1
  let k = 0
  while k < n do
    let rid = to_num(rows[k][0])
    if rid > max_id then
      let max_id = rid
    end
    let k = k + 1
  end
  if max_id + 1 > q_get_counter(topic) then
    q_set_counter(topic, max_id + 1)
  end
  return true
end

# queue_publish(topic, payload) -> msg_id (as a string). Appends a new
# "pending" message.
make a function called queue_publish takes topic, payload returns msg_id
  q_migrate_if_old(topic)
  let new_id = q_get_counter(topic)
  q_put_message(topic, new_id, "pending", payload)
  q_set_counter(topic, new_id + 1)
  return new_id
end

# queue_consume(topic) -> payload of the oldest pending message, or "" if
# none pending. Does NOT ack -- call queue_ack once the message has
# genuinely been handled, so a crash between consume and ack leaves it
# recoverable rather than silently lost.
make a function called queue_consume takes topic returns payload
  let rows = q_read(topic)
  let i = 0
  let n = to_num(list_len(rows))
  while i < n do
    let row = rows[i]
    if row[1] == "pending" then
      return row[2]
    end
    let i = i + 1
  end
  return ""
end

# queue_ack(topic, msg_id) -> marks the given message acked (rewrites
# only that one message's own file/key, not the whole topic).
make a function called queue_ack takes topic, msg_id returns done
  q_migrate_if_old(topic)
  let padded = q_pad(to_num(msg_id))
  let existing = q_message_ids(topic)
  if q_list_contains(existing, padded) == false then
    return true
  end
  let unpacked = q_unpack_status_payload(q_read_message(topic, padded))
  q_put_message(topic, to_num(msg_id), "acked", unpacked[1])
  return true
end

# queue_pending(topic) -> list of message ids still pending. What makes the
# queue "resumable": on a fresh process, this is simply whatever is still
# unacked, read straight from storage -- no separate load/restore step.
make a function called queue_pending takes topic returns ids
  let rows = q_read(topic)
  let out = []
  let i = 0
  let n = to_num(list_len(rows))
  while i < n do
    let row = rows[i]
    if row[1] == "pending" then
      let out = list_push(out, row[0])
    end
    let i = i + 1
  end
  return out
end

# queue_remove_topic(topic) -> done. Fully removes every message and the
# topic's own counter, so the next publish starts completely fresh (ids
# restart at 0). For test/spec fixture resets: truncating or deleting the
# old q_file(topic) path no longer has any effect once a topic is in the
# new per-message layout, so anything that used to reset a topic that way
# should call this instead.
make a function called queue_remove_topic takes topic returns done
  q_migrate_if_old(topic)
  if queue_driver() == "vfs" then
    if vfs_exists(q_key(topic)) == "1" then
      vfs_delete(q_key(topic))
    end
    let keys = vfs_list(q_msg_prefix(topic))
    let i = 0
    while i < list_len(keys) do
      vfs_delete(keys[i])
      let i = i + 1
    end
    return true
  end
  if file_exists(q_file(topic)) == true then
    remove_file(q_file(topic))
  end
  if file_exists(q_msg_dir(topic)) == true then
    let entries = list_dir(q_msg_dir(topic))
    let i = 0
    while i < list_len(entries) do
      remove_file(q_msg_dir(topic) + "/" + entries[i])
      let i = i + 1
    end
    remove_file(q_msg_dir(topic))
  end
  return true
end

# queue_list_topics() -> every topic name with any content, old or new
# format, de-duplicated. Replaces the old idiom of a caller scanning
# list_dir(q_dir()) and stripping ".log" itself (task_registry.patlang
# used to do exactly that -- a direct dependency on this module's own
# on-disk shape that this migration would otherwise have broken).
# list_dir suffixes directory entries with "/" (hosts.rs's own host_
# list_dir), which is what distinguishes a new-format topic directory
# from an old, not-yet-migrated ".log" file here.
make a function called queue_list_topics returns topics
  if queue_driver() == "vfs" then
    return q_list_topics_vfs()
  end
  if file_exists(q_dir()) != true then
    return []
  end
  let entries = list_dir(q_dir())
  let out = []
  let i = 0
  let n = to_num(list_len(entries))
  while i < n do
    let entry = entries[i]
    if str_ends_with(entry, "/") then
      let out = list_push(out, substr(entry, 0, entry.length - 1))
    elif str_ends_with(entry, ".log") then
      let out = list_push(out, substr(entry, 0, entry.length - 4))
    end
    let i = i + 1
  end
  return out
end

make a function called q_list_topics_vfs returns topics
  let keys = vfs_list("/queue/")
  let seen = []
  let out = []
  let i = 0
  while i < list_len(keys) do
    let rel = substr(keys[i], 7, keys[i].length - 7)
    let topic = ""
    if str_ends_with(rel, ".log") then
      let topic = substr(rel, 0, rel.length - 4)
    else
      let slash = find_substr_queue(rel, "/")
      if slash >= 0 then
        let topic = substr(rel, 0, slash)
      end
    end
    if (topic != "") and (q_list_contains(seen, topic) == false) then
      let seen = list_push(seen, topic)
      let out = list_push(out, topic)
    end
    let i = i + 1
  end
  return out
end

# ---- multi-topic reads, for the "give every writer its own topic" idiom ----
#
# PatLang has no locking primitive, so a topic shared between multiple
# concurrent writers can still race on id allocation (unchanged by the
# per-message storage above -- see this file's own header). The idiomatic
# fix isn't locking -- it's giving each writer its own topic name (e.g.
# queue_writer_topic below: a base name plus the writer's own identity) so
# no two writers ever touch the same message set at all. A reader that
# needs the combined stream then reads across that whole family of topics
# with the functions below, rather than one consumer trying to read one
# shared, contended topic.

# queue_writer_topic(base, writer_id) -> a topic name namespaced to one
# writer, e.g. queue_writer_topic("orders", "worker_1") -> "orders__worker_1".
# Purely a naming convention -- any unique-per-writer string works, this
# just gives it one canonical shape so producers and consumers agree.
make a function called queue_writer_topic takes base, writer_id returns topic
  return base + "__" + writer_id
end

# queue_pending_multi(topics) -> list of [topic, msg_id] pairs, one per
# still-pending message across every topic in the given list, in the order
# the topics were given (and id order within each topic).
make a function called queue_pending_multi takes topics returns pairs
  let out = []
  let t = 0
  let tn = to_num(list_len(topics))
  while t < tn do
    let topic = topics[t]
    let ids = queue_pending(topic)
    let i = 0
    let n = to_num(list_len(ids))
    while i < n do
      let out = list_push(out, [topic, ids[i]])
      let i = i + 1
    end
    let t = t + 1
  end
  return out
end

# queue_consume_multi(topics) -> [topic, payload] for the oldest pending
# message in the FIRST topic (in list order) that has one; ["", ""] if none
# of the given topics have anything pending. Simple "check topics in order"
# semantics -- there's no id shared across topics, so no attempt is made to
# merge them into one global time order.
make a function called queue_consume_multi takes topics returns result
  let t = 0
  let tn = to_num(list_len(topics))
  while t < tn do
    let topic = topics[t]
    let payload = queue_consume(topic)
    if payload != "" then
      return [topic, payload]
    end
    let t = t + 1
  end
  return ["", ""]
end

# queue_attempt(topic, payload, action) -> result of action(). Publishes an
# intent message before calling action, acks it once action returns
# normally. If action aborts (fatal host error -- PatLang has no
# try/catch, this doesn't change that), the intent message stays
# "pending" on disk for a later run to find via queue_pending and retry
# or compensate.
make a function called queue_attempt takes topic, payload, action returns result
  let msg_id = queue_publish(topic, payload)
  let result = action()
  queue_ack(topic, msg_id)
  return result
end

# Build-time transcript form of queue_recovery_demo.patlang, for the site
# demo page. A real crash is a real fatal process exit (see the genuine,
# actually-run-it-yourself form for that -- self_hosting/examples/
# queue_recovery_demo.patlang, verified this session across four real,
# separate OS processes: worker_1 and worker_2 completing normally,
# worker_3 genuinely crashing mid-attempt on a real fatal read_file error,
# and a fourth "recover" process finding and completing worker_3's
# leftover checkpoint). A fatal error can't be captured mid-transcript
# here without killing this whole build, so worker_3's crash is
# represented directly: its intent is published (exactly what
# queue_attempt does before calling the risky action) without ever
# calling the action or the ack, which is precisely the on-disk state a
# real crash between those two steps leaves behind.

make a function called run_queue_recovery_transcript returns done
  qrt_reset()
  print("=== phase 1: three independent workers, each with its own topic ===")

  let ok1 = || do
    print("worker_1: did the work successfully")
    return true
  end
  queue_attempt(queue_writer_topic(qrt_base(), "worker_1"), "job for worker_1", ok1)
  print("worker_1: job completed and acked")

  let ok2 = || do
    print("worker_2: did the work successfully")
    return true
  end
  queue_attempt(queue_writer_topic(qrt_base(), "worker_2"), "job for worker_2", ok2)
  print("worker_2: job completed and acked")

  print("worker_3: starting risky work...")
  print("worker_3: CRASH -- fatal host error mid-attempt (verified for real: a bad read_file call kills the process before it reaches queue_ack)")
  queue_publish(queue_writer_topic(qrt_base(), "worker_3"), "job for worker_3")
  print("(process exits here for real; only the intent message is left on disk)")

  print("")
  print("=== phase 2: a separate, later process recovers ===")
  let topics = qrt_topics()
  let pending = queue_pending_multi(topics)
  let pn = to_num(list_len(pending))
  print("RECOVERY: " + pn + " leftover pending message(s) across 3 worker topic(s)")
  let i = 0
  while i < pn do
    let pair = pending[i]
    let topic = pair[0]
    let msg_id = pair[1]
    let payload = queue_consume(topic)
    print("RECOVERY: found interrupted job -- " + payload + " (topic " + topic + ")")
    print("RECOVERY: completing it now...")
    queue_ack(topic, msg_id)
    print("RECOVERY: acked, done")
    let i = i + 1
  end

  print("")
  print("=== phase 3: recovery is idempotent -- run it again, nothing left ===")
  let pending2 = queue_pending_multi(topics)
  print("RECOVERY: " + to_num(list_len(pending2)) + " leftover pending message(s)")

  return true
end
